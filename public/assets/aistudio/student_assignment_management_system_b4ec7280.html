โค้ด Google Apps Script สำหรับเชื่อมต่อ Google Sheet

1. เปิด Google Sheet ใหม่ แล้วไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
2. คัดลอกโค้ด JavaScript ด้านล่างนี้ไปวางในเครื่องมือ Apps Script แล้วกดบันทึก
3. กดปุ่ม ทำให้ใช้งานได้ (Deploy) > การทําให้ใช้งานได้ใหม่ (New deployment)
4. เลือกประเภทเป็น เว็บแอป (Web app) กำหนดให้ "ผู้ที่มีลิงก์ (Anyone)" เข้าถึงได้
5. คัดลอก Web App URL นำมากรอกในช่องตั้งค่าระบบ 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (sheet.getLastRow() == 0) {
      sheet.appendRow(["Timestamp", "Student ID", "Student Name", "Subject", "Assignment Title", "Score", "Max Score", "Status", "Links/Files"]);
    }
    
    sheet.appendRow([
      new Date(),
      data.studentId,
      data.studentName,
      data.subject,
      data.assignmentTitle,
      data.score || "",
      data.maxScore || "",
      data.status || "ส่งแล้ว",
      data.attachments || data.link || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
